# Independent remote-only bootstrap tracer. Never included by production CMake.
cmake_minimum_required(VERSION 3.24)
project(GswBootstrapFixture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This explicitly test-only tracer requires Windows/MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GSW_BOOTSTRAP_FIXTURE_ROOT "D:/gb/owned" CACHE PATH "Only owned remote fixture destinations are accepted")
add_executable(gsw_bootstrap_fixture ../tests/NativeBootstrapFixtureMain.cpp)
add_executable(gsw_bootstrap_terminal_harness ../tests/NativeBootstrapTerminalHarness.cpp)
add_executable(gsw_bootstrap_loader_probe ../tests/NativeBootstrapLoaderProbe.cpp)
foreach(target gsw_bootstrap_fixture gsw_bootstrap_terminal_harness gsw_bootstrap_loader_probe)
  target_include_directories(${target} PRIVATE ../tests ../third_party/nlohmann)
  target_compile_definitions(${target} PRIVATE GSW_BOOTSTRAP_FIXTURE_ONLY=1
    GSW_BOOTSTRAP_FIXTURE_ROOT="${GSW_BOOTSTRAP_FIXTURE_ROOT}" NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
  target_link_libraries(${target} PRIVATE advapi32 user32 bcrypt wintrust crypt32)
endforeach()
set_target_properties(gsw_bootstrap_fixture PROPERTIES OUTPUT_NAME "GSW Bootstrap Fixture")
set_target_properties(gsw_bootstrap_terminal_harness PROPERTIES OUTPUT_NAME "GSW Bootstrap Terminal Harness")
set_target_properties(gsw_bootstrap_loader_probe PROPERTIES OUTPUT_NAME "GSW Bootstrap Loader Probe")
# Deliberately no install() or dependency from the public application/helper.
